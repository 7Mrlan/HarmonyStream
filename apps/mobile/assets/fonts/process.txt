3. JavaScript 动画与交互逻辑
// 元素引用
const svg = document.getElementById('canvas');
const ship = document.getElementById('ship-container');
const interactionZone = document.getElementById('interaction-zone');
const progressFill = document.getElementById('progress-fill');
const chargeGroup = document.getElementById('charge-particles');
const starDustGroup = document.getElementById('star-dust');
const coordDisplay = document.getElementById('coord-display');
const velocityDisplay = document.getElementById('velocity-display');

// 状态变量
let isDragging = false;
let progress = 0.4966; // 初始位置对应图一 49.66%
let targetProgress = progress;
let velocity = 0;
let lastX = 0;
let lastTime = Date.now();

// 获取本地 X 坐标（相对于进度条）
function getLocalX(e) {
    const CTM = svg.getScreenCTM();
    const pt = svg.createSVGPoint();
    pt.x = e.clientX;
    pt.y = e.clientY;
    const localPt = pt.matrixTransform(CTM.inverse());
    return Math.max(0, Math.min(700, localPt.x - 50));
}

// 创建流动电荷粒子
function createCharges() {
    for (let i = 0; i < 30; i++) {
        const rect = document.createElementNS("http://www.w3.org/2000/svg", "rect");
        rect.setAttribute("width", "3");
        rect.setAttribute("height", "3");
        rect.setAttribute("fill", "var(--cyan)");
        chargeGroup.appendChild(rect);
        resetCharge(rect);
    }
}

function resetCharge(rect) {
    const x = Math.random() * 700;
    const y = 10 + Math.random() * 40;
    gsap.set(rect, { x, y, opacity: Math.random() * 0.7 });
    gsap.to(rect, {
        x: "-=60",
        duration: 0.4 + Math.random() * 0.6,
        repeat: -1,
        ease: "none",
        onRepeat: () => {
            if (parseFloat(rect.getAttribute('x')) < 0) {
                gsap.set(rect, { x: 700 });
            }
        }
    });
}

// 生成星尘残影
function spawnDust(x) {
    const dust = document.createElementNS("http://www.w3.org/2000/svg", "rect");
    const size = 4 + Math.random() * 6;
    dust.setAttribute("width", size);
    dust.setAttribute("height", size);
    dust.setAttribute("fill", Math.random() > 0.5 ? "var(--purple)" : "var(--cyan)");
    dust.setAttribute("filter", "url(#pixel-glow)");
    starDustGroup.appendChild(dust);

    gsap.set(dust, { x: x, y: 20 + Math.random() * 20, opacity: 0.8 });
    gsap.to(dust, {
        x: x - 40 - Math.random() * 60,
        y: "+=" + (Math.random() * 20 - 10),
        opacity: 0,
        duration: 0.8,
        ease: "power2.out",
        onComplete: () => dust.remove()
    });
}

// 交互事件
interactionZone.addEventListener('pointerdown', (e) => {
    isDragging = true;
    interactionZone.setPointerCapture(e.pointerId);
    updateFromEvent(e);
    ship.style.cursor = 'grabbing';
});

interactionZone.addEventListener('pointermove', (e) => {
    if (!isDragging) return;
    updateFromEvent(e);
});

interactionZone.addEventListener('pointerup', () => {
    isDragging = false;
    ship.style.cursor = 'grab';
});

function updateFromEvent(e) {
    const localX = getLocalX(e);
    targetProgress = localX / 700;
    
    const now = Date.now();
    const dt = (now - lastTime) / 1000;
    if (dt > 0) {
        velocity = Math.abs(localX - lastX) / dt;
        lastX = localX;
        lastTime = now;
    }

    if (velocity > 100) {
        spawnDust(localX);
    }
}

// 主动画循环
function update() {
    progress += (targetProgress - progress) * 0.15;
    const displayX = progress * 700;
    
    gsap.set(ship, { x: displayX });
    progressFill.setAttribute("width", displayX);

    // 飞船动态变形
    const scaleX = 1 + (velocity / 5000);
    const skewY = (targetProgress - progress) * 50;
    gsap.set("#ship-wrapper", { 
        scaleX: scaleX, 
        skewY: skewY
    });

    // 更新 UI
    coordDisplay.textContent = `X: ${(progress * 100).toFixed(2)}%`;
    velocityDisplay.textContent = `${velocity.toFixed(2)} PX/S`;

    if (!isDragging) {
        velocity *= 0.95;
        if (velocity < 0.1) velocity = 0;
    }

    requestAnimationFrame(update);
}

// 初始化
createCharges();
update();
